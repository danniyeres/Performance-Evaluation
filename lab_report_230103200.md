## Shared harness source

```python
"""Shared timing and verification helpers for the performance lab."""
from dataclasses import dataclass
import multiprocessing as mp
import time
from typing import Callable


@dataclass
class BenchStats:
    min_ms: float
    median_ms: float
    iqr_ms: float


def fnv1a_hash(value: str) -> int:
    h = 14695981039346656037
    for byte in value.encode("utf-8"):
        h = ((h ^ byte) * 1099511628211) & 0xFFFFFFFFFFFFFFFF
    return h


def run_benchmark(fn: Callable[[], None], warmup: int = 2, runs: int = 7) -> BenchStats:
    for _ in range(warmup):
        fn()
    samples = []
    for _ in range(runs):
        start = time.perf_counter_ns()
        fn()
        samples.append((time.perf_counter_ns() - start) / 1e6)
    samples.sort()
    return BenchStats(samples[0], samples[len(samples) // 2],
                      samples[(3 * len(samples)) // 4] - samples[len(samples) // 4])


def emit_verify_tag(phase: str, seed: int, m1: float, m2: float) -> str:
    hw = mp.cpu_count()
    sig = fnv1a_hash(f"{phase}|{seed}|{hw}|{m1:.4f}|{m2:.4f}")
    return f"VERIFY_HASH[{phase}]: {sig:016X} (P_log={hw})"
```

### task1_falsesharing.py

```python
"""Phase 1: shared-cache-line stride sweep and process scaling."""
import ctypes
import multiprocessing as mp
from multiprocessing import shared_memory
from harness import run_benchmark, emit_verify_tag

STUDENT_ID = 230103200
M = 16_000_000
MASK = (1 << 64) - 1


def fs_worker(shm_name: str, slot_idx: int, iters: int) -> None:
    shm = shared_memory.SharedMemory(name=shm_name)
    arr = (ctypes.c_uint64 * 512).from_buffer(shm.buf)
    acc = arr[slot_idx]
    for i in range(iters):
        acc = (acc * 6364136223846793005 + (i | 1)) & MASK
        arr[slot_idx] = acc
    del arr
    shm.close()


def trial(p: int, stride: int, total: int = M) -> None:
    shm = shared_memory.SharedMemory(create=True, size=4096)
    workers = []
    try:
        slots = (ctypes.c_uint64 * 512).from_buffer(shm.buf)
        for i in range(p):
            slots[i * stride] = i + 1
        del slots
        base, rem = divmod(total, p)
        for i in range(p):
            proc = mp.Process(target=fs_worker,
                              args=(shm.name, i * stride, base + (i < rem)))
            workers.append(proc)
        for proc in workers:
            proc.start()
        for proc in workers:
            proc.join()
            if proc.exitcode:
                raise RuntimeError(f"worker exited with status {proc.exitcode}")
    finally:
        shm.close()
        shm.unlink()


def main() -> None:
    pphys = 8
    print("stride sweep: stride, median_ms, IQR_ms, Mops/s")
    sweep = {}
    for stride in (1, 2, 4, 8, 16, 32):
        stats = run_benchmark(lambda: trial(pphys, stride), warmup=0, runs=3)
        sweep[stride] = stats.median_ms
        print(f"{stride}, {stats.median_ms:.3f}, {stats.iqr_ms:.3f}, {M/stats.median_ms/1000:.3f}")
    print("scaling: p, stride, median_ms, speedup, efficiency, karp_flatt")
    for stride in (1, 16):
        t1 = None
        for p in (1, 2, 4, 8):
            stats = run_benchmark(lambda p=p, stride=stride: trial(p, stride), warmup=0, runs=3)
            t1 = stats.median_ms if p == 1 else t1
            speedup = t1 / stats.median_ms
            e = (1/speedup - 1/p) / (1 - 1/p) if p > 1 else 0.0
            print(f"{p}, {stride}, {stats.median_ms:.3f}, {speedup:.3f}, {speedup/p:.3f}, {e:.5f}")
    print(emit_verify_tag("PHASE1", STUDENT_ID, sweep[1], sweep[16]))


if __name__ == "__main__":
    main()
```

### task2_scaling.py

```python
"""Phase 2: strong and weak scaling using a persistent process pool."""
import math
import multiprocessing as mp
import time
from harness import run_benchmark, emit_verify_tag

STUDENT_ID = 230103200
N_STRONG = 1_200_000
N0 = 200_000


def stage_a(n: int) -> float:
    acc = 0.0
    for i in range(n):
        x = i * 1e-6 + 0.1
        for _ in range(10):
            acc += math.sin(x) * math.sqrt(abs(x) + 1.0)
    return acc


def stage_b(bounds: tuple[int, int]) -> float:
    lo, hi = bounds
    acc = 0.0
    for i in range(lo, hi):
        x = i * 1e-6 + 0.1
        for _ in range(30):
            acc += math.sin(x) * math.cos(x) + math.sqrt(abs(x) + 1.0) * 1e-3
    return acc


def pipeline(pool, n: int, p: int) -> None:
    stage_a(max(1, int(n * 0.05)))
    q, r = divmod(n, p)
    bounds = []
    lo = 0
    for i in range(p):
        hi = lo + q + (i < r)
        bounds.append((lo, hi))
        lo = hi
    pool.map(stage_b, bounds)


def time_stage_a(n: int) -> float:
    start = time.perf_counter_ns()
    stage_a(n)
    return (time.perf_counter_ns() - start) / 1e6


def main() -> None:
    maxp = mp.cpu_count()
    strong = {}
    with mp.Pool(maxp) as pool:
        for p in range(1, maxp + 1):
            stats = run_benchmark(lambda p=p: pipeline(pool, N_STRONG, p), warmup=0, runs=3)
            strong[p] = stats.median_ms
            print(f"strong p={p}: {stats.median_ms:.3f} ms speedup={strong[1]/stats.median_ms:.3f} efficiency={strong[1]/stats.median_ms/p:.3f}")
        a0 = time_stage_a(max(1, int(N_STRONG * .05)))
        s = a0 / strong[1]
        print(f"Amdahl serial fraction={s:.5f}; cap={1/s:.3f}")
        weak = {}
        gustafson = {}
        for p in range(1, maxp + 1):
            stats = run_benchmark(lambda p=p: pipeline(pool, p * N0, p), warmup=0, runs=3)
            weak[p] = stats.median_ms
            alpha = time_stage_a(max(1, int(p * N0 * .05))) / weak[p]
            scaled_speedup = p - alpha * (p - 1)
            gustafson[p] = scaled_speedup
            print(f"weak p={p}: {stats.median_ms:.3f} ms efficiency={weak[1]/stats.median_ms:.3f} Gustafson={scaled_speedup:.3f}")
    print(emit_verify_tag("PHASE2", STUDENT_ID,
                          strong[maxp] and strong[1]/strong[maxp],
                          gustafson[maxp]))


if __name__ == "__main__":
    main()
```

### task3_roofline_gemm.py

```python
"""Phase 3: memory/compute ceilings and three GEMM loop organizations."""
import multiprocessing as mp
from multiprocessing import shared_memory
import ctypes
import time
from harness import run_benchmark, emit_verify_tag

STUDENT_ID = 230103200
N = 256
PPHYS = 8


def copy_worker(args):
    name_a, name_b, lo, hi = args
    a = shared_memory.SharedMemory(name=name_a)
    b = shared_memory.SharedMemory(name=name_b)
    av, bv = a.buf, b.buf
    av[lo:hi] = bv[lo:hi]
    del av, bv
    a.close(); b.close()


def fma_worker(rounds):
    x = [float(i + 1) / 17 for i in range(16)]
    y = [float(i + 3) / 19 for i in range(16)]
    for _ in range(rounds):
        x[0] = x[0] * y[0] + 1.0; x[1] = x[1] * y[1] + 1.0
        x[2] = x[2] * y[2] + 1.0; x[3] = x[3] * y[3] + 1.0
        x[4] = x[4] * y[4] + 1.0; x[5] = x[5] * y[5] + 1.0
        x[6] = x[6] * y[6] + 1.0; x[7] = x[7] * y[7] + 1.0
        x[8] = x[8] * y[8] + 1.0; x[9] = x[9] * y[9] + 1.0
        x[10] = x[10] * y[10] + 1.0; x[11] = x[11] * y[11] + 1.0
        x[12] = x[12] * y[12] + 1.0; x[13] = x[13] * y[13] + 1.0
        x[14] = x[14] * y[14] + 1.0; x[15] = x[15] * y[15] + 1.0
    return sum(x)


def roofline():
    size = 64 * 1024 * 1024
    a = shared_memory.SharedMemory(create=True, size=size)
    b = shared_memory.SharedMemory(create=True, size=size)
    try:
        b.buf[:] = b"x" * size
        q, r = divmod(size, PPHYS)
        chunks = []
        lo = 0
        for i in range(PPHYS):
            hi = lo + q + (i < r)
            chunks.append((a.name, b.name, lo, hi))
            lo = hi
        with mp.Pool(PPHYS) as pool:
            copy = run_benchmark(lambda: pool.map(copy_worker, chunks), warmup=0, runs=3)
            bandwidth = 2 * size / (copy.median_ms * 1e6)
            rounds = 60_000_000 // 32
            compute = run_benchmark(lambda: pool.map(fma_worker, [rounds] * PPHYS), warmup=0, runs=3)
            gflops = 60_000_000 * PPHYS / (compute.median_ms * 1e6)
        print(f"DRAM copy: {copy.median_ms:.3f} ms; {bandwidth:.3f} GB/s")
        print(f"Scalar compute: {compute.median_ms:.3f} ms; {gflops:.3f} GFLOP/s; ridge={gflops/bandwidth:.3f} FLOP/B")
    finally:
        a.close(); a.unlink(); b.close(); b.unlink()


def gemm_naive(a, b, n):
    c = [0.0] * (n*n)
    for i in range(n):
        for j in range(n):
            x = 0.0
            for k in range(n):
                x += a[i*n+k] * b[k*n+j]
            c[i*n+j] = x
    return c


def gemm_ikj(a, b, n):
    c = [0.0] * (n*n)
    for i in range(n):
        for k in range(n):
            aik = a[i*n+k]
            for j in range(n):
                c[i*n+j] += aik * b[k*n+j]
    return c


def gemm_blocked(a, b, n, bs):
    c = [0.0] * (n*n)
    for ii in range(0, n, bs):
        for kk in range(0, n, bs):
            for jj in range(0, n, bs):
                for i in range(ii, min(ii+bs, n)):
                    for k in range(kk, min(kk+bs, n)):
                        aik = a[i*n+k]
                        row = i*n
                        for j in range(jj, min(jj+bs, n)):
                            c[row+j] += aik * b[k*n+j]
    return c


def gemm_rows(args):
    a, b, n, first, last, variant, bs = args
    c = [0.0] * ((last - first) * n)
    if variant == "blocked":
        for ii in range(first, last, bs):
            for kk in range(0, n, bs):
                for jj in range(0, n, bs):
                    for i in range(ii, min(ii+bs, last)):
                        row = (i - first) * n
                        for k in range(kk, min(kk+bs, n)):
                            aik = a[i*n+k]
                            for j in range(jj, min(jj+bs, n)):
                                c[row+j] += aik * b[k*n+j]
        return first, c
    for i in range(first, last):
        row = (i - first) * n
        if variant == "naive":
            for j in range(n):
                total = 0.0
                for k in range(n):
                    total += a[i*n+k] * b[k*n+j]
                c[row+j] = total
        elif variant == "ikj":
            for k in range(n):
                aik = a[i*n+k]
                for j in range(n):
                    c[row+j] += aik * b[k*n+j]
    return first, c


def parallel_gemm(pool, a, b, n, variant, bs=None):
    q, r = divmod(n, PPHYS)
    chunks = []
    first = 0
    for worker in range(PPHYS):
        last = first + q + (worker < r)
        chunks.append((a, b, n, first, last, variant, bs))
        first = last
    output = [0.0] * (n*n)
    for first, part in pool.map(gemm_rows, chunks):
        output[first*n:(first+len(part)//n)*n] = part
    return output


def main():
    roofline()
    n = N
    a = [((i % 31) - 15) / 17 for i in range(n*n)]
    b = [((i % 29) - 14) / 13 for i in range(n*n)]
    stats = {}
    outputs = {}
    best = None
    with mp.Pool(PPHYS) as pool:
        for name, variant in (("naive", "naive"), ("ikj", "ikj")):
            fn = lambda variant=variant: parallel_gemm(pool, a, b, n, variant)
            bench = run_benchmark(fn, warmup=0, runs=3)
            stats[name] = bench.median_ms
            outputs[name] = fn()
            print(f"{name}: {bench.median_ms:.3f} ms, {2*n**3/(bench.median_ms*1e6):.4f} GFLOP/s")
        for bs in (8,16,32,64,96,128):
            fn = lambda bs=bs: parallel_gemm(pool, a, b, n, "blocked", bs)
            bench = run_benchmark(fn, warmup=0, runs=3)
            result = fn()
            err = max(abs(x-y) for x,y in zip(result, outputs["ikj"]))
            rate = 2*n**3/(bench.median_ms*1e6)
            print(f"BS={bs}: {bench.median_ms:.3f} ms, {rate:.4f} GFLOP/s, max_error={err:.3g}, Wtile={3*bs*bs*8} B, I≈{bs/12:.3f}")
            if best is None or bench.median_ms < best[0]:
                best = (bench.median_ms, rate, bs)
```

```python
"""Phase 4: static and dynamic scheduling for a skewed workload."""
import math
import multiprocessing as mp
import time
from harness import emit_verify_tag

STUDENT_ID = 230103200
N = 8000


def work(i: int) -> float:
    acc = 0.0
    for j in range(1 + int((i / 40) ** 2)):
        x = (i + j) * 1e-4
        acc += math.sin(x) * math.cos(x) + math.sqrt(abs(x) + 1.0)
    return acc


def worker_static(items, timings, slot):
    start = time.perf_counter()
    for i in items:
        work(i)
    timings[slot] = time.perf_counter() - start


def worker_dynamic(counter, lock, timings, slot, chunk):
    start = time.perf_counter()
    while True:
        with lock:
            lo = counter.value
            counter.value += chunk
        if lo >= N:
            break
        for i in range(lo, min(N, lo + chunk)):
            work(i)
    timings[slot] = time.perf_counter() - start


def run_static(p, cyclic=False):
    timings = mp.Array('d', p)
    procs = []
    for rank in range(p):
        items = range(rank, N, p) if cyclic else range(rank*N//p, (rank+1)*N//p)
        procs.append(mp.Process(target=worker_static, args=(items, timings, rank)))
    start=time.perf_counter()
    for x in procs: x.start()
    for x in procs: x.join()
    wall=time.perf_counter()-start
    values=list(timings)
    imbalance=max(values)/(sum(values)/p)
    return wall, imbalance


def run_dynamic(p, chunk):
    counter=mp.Value('i', 0)
    lock=mp.Lock()
    timings=mp.Array('d', p)
    procs=[mp.Process(target=worker_dynamic,args=(counter,lock,timings,i,chunk)) for i in range(p)]
    start=time.perf_counter()
    for x in procs: x.start()
    for x in procs: x.join()
    wall=time.perf_counter()-start
    values=list(timings)
    return wall, max(values)/(sum(values)/p)


def main():
    p=mp.cpu_count()
    block, lb=run_static(p)
    cyc, lc=run_static(p, True)
    print(f"block: {block:.3f}s lambda={lb:.3f}; cyclic: {cyc:.3f}s lambda={lc:.3f}")
    results={}
    for c in (1,8,64,512):
        results[c]=run_dynamic(p,c)
        print(f"dynamic C={c}: {results[c][0]:.3f}s lambda={results[c][1]:.3f}")
    best=min(results,key=lambda c:results[c][0])
    print(f"selected C*={best}")
    print(emit_verify_tag("PHASE4", STUDENT_ID, block*1000, results[best][0]*1000))


if __name__ == "__main__": main()
```
